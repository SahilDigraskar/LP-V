 pragma solidity ^0.8.0;

 contract Bank {

     address public manager;
     uint256 private balance = 0;

     constructor() {
         manager = msg.sender;
     }

     function deposit() public payable {
         require(
             msg.value > 0,
             "Deposit amount should be greater than 0."
         );

         balance += msg.value;
     }

     function showBalance() public view returns (uint256) {
         return balance;
     }
 }
